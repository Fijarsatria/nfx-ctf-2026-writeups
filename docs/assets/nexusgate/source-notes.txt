  if (req.url.startsWith('/internal/')) {
    res.writeHead(403, { 'content-type': 'text/plain' });
    res.end('forbidden');
    return;
  }

  // ---- 2. ticket pages: edge-cached fetch ----
  if (req.method === 'GET' && req.url.startsWith('/tickets/')) {
    ticketDispatcher.request({
      origin: BACKEND_ORIGIN,
      path: req.url,
      method: req.method,
      headers: rebuildHeaders(req),
    }).then((r) => {
      res.writeHead(r.statusCode, r.headers);
      r.body.pipe(res);
    }).catch(() => {
      res.writeHead(502, { 'content-type': 'text/plain' });
      res.end('bad gateway');
    });
    return;
  }

  // ---- 3. generic forwarding: rebuild headers + pipe body ----
  const headers = rebuildHeaders(req);
  // explicit framing for requests with no visible body indicator
  if (!req.headers['content-length'] && !req.headers['transfer-encoding']) {
    headers['content-length'] = '0';

'use strict';
const { DatabaseSync } = require('node:sqlite');
const crypto = require('crypto');

const ADMIN_PASSWORD = crypto.randomBytes(16).toString('hex');
const PHASE_TOKEN = crypto.randomBytes(16).toString('hex');
const MASTER_TICKET = crypto.randomBytes(16).toString('hex');

// In-memory identity and vault store
const db = new DatabaseSync(':memory:');
db.exec(`
  CREATE TABLE users (username TEXT PRIMARY KEY, password TEXT);
  CREATE TABLE vault (owner TEXT, secret TEXT);
  CREATE TABLE journal (username TEXT, entry TEXT);
`);

db.prepare('INSERT INTO users VALUES (?, ?)').run('admin', ADMIN_PASSWORD);
db.prepare('INSERT INTO vault VALUES (?, ?)').run('admin', PHASE_TOKEN);

// Tagged template prepared statement manager
const tagStore = db.createTagStore();

// Admin telemetry heartbeat sync
setInterval(() => {
  try {
    tagStore.all`SELECT * FROM vault WHERE owner = ${'admin'}`;
  } catch (_) { /* heartbeat error suppressed */ }
}, 10000);

// Background worker for periodic audit validation
const botCookie = 'session=' + crypto.randomBytes(8).toString('hex');
sessions.set(botCookie.slice(8), 'admin');

function botCycle() {
  const nonce = crypto.randomBytes(8).toString('hex');
  const req = http.get({
    host: '127.0.0.1',
    port: 80,
    path: `/tickets/latest?aud=${nonce}`,
    headers: {
      cookie: botCookie,
      'user-agent': 'nx-audit-bot/1.0',
      accept: 'text/html,application/xhtml+xml',
    },
  }, (res) => {
    res.resume();
    res.on('end', () => {
      try { fs.writeFileSync('/app/data/bot', nonce); } catch (_) { /* write error suppressed */ }
    });
  });
  req.on('error', () => { /* connection error suppressed */ });
}

function startBotWorker() {
  setInterval(botCycle, 45000);
}
