#!/usr/bin/env bash
set -euo pipefail

mkdir -p /out /var/www/uploads /var/automation/exports /var/automation/data
chown -R www-data:www-data /out /var/www/uploads
chmod 2775 /var/www/uploads
chmod 0775 /out

start_as() {
  local user="$1"
  shift
  local uid gid
  uid="$(id -u "$user")"
  gid="$(id -g "$user")"
  setpriv --reuid="$uid" --regid="$gid" --clear-groups -- "$@" &
}

python3 -u /opt/automation/app.py &
start_as svc-watch python3 -u /opt/watchtower/app.py
